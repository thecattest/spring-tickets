     ```bash
     python -c "
     from database import engine
     with engine.connect() as conn:
         # SQLite doesn't support ALTER COLUMN, but does support creating new nullable columns.
         # However, these columns already exist as NOT NULL. SQLite workaround:
         # Check if columns are already nullable by attempting an INSERT with NULL.
         # If they are NOT NULL, we need to recreate the table.
         import sqlite3
         raw = conn.connection.connection  # unwrap to raw sqlite3 connection
         cursor = raw.cursor()
         # Get current table info
         cursor.execute('PRAGMA table_info(payments)')
         columns = {row[1]: row[3] for row in cursor.fetchall()}  # name -> notnull
         vk_url_notnull = columns.get('vk_message_url', 0)
         vk_ts_notnull = columns.get('vk_message_ts', 0)
         if vk_url_notnull or vk_ts_notnull:
             # Need to recreate table with nullable columns
             # Since payments table may be empty in dev, just recreate
             cursor.execute('SELECT COUNT(*) FROM payments')
             count = cursor.fetchone()[0]
             if count == 0:
                 cursor.execute('DROP TABLE IF EXISTS payment_seats')
                 cursor.execute('DROP TABLE IF EXISTS payments')
                 raw.commit()
                 print('Dropped empty payments tables for recreation')
             else:
                 # Has data - use SQLite table rebuild pattern
                 cursor.execute('PRAGMA foreign_keys=OFF')
                 cursor.execute('BEGIN TRANSACTION')
                 cursor.execute('''CREATE TABLE payments_new (
                     id INTEGER PRIMARY KEY AUTOINCREMENT,
                     reservation_id INTEGER NOT NULL REFERENCES reservations(id),
                     recorded_by_user_id INTEGER NOT NULL REFERENCES app_users(id),
                     vk_message_url TEXT,
                     vk_message_ts DATETIME,
                     payment_amount NUMERIC(10,2) NOT NULL,
                     seat_count INTEGER NOT NULL,
                     recorded_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                     note TEXT
                 )''')
                 cursor.execute('INSERT INTO payments_new SELECT * FROM payments')
                 cursor.execute('DROP TABLE payments')
                 cursor.execute('ALTER TABLE payments_new RENAME TO payments')
                 cursor.execute('CREATE INDEX payments_reservation_idx ON payments(reservation_id)')
                 cursor.execute('CREATE INDEX payments_vk_message_ts_idx ON payments(vk_message_ts)')
                 cursor.execute('COMMIT')
                 cursor.execute('PRAGMA foreign_keys=ON')
                 raw.commit()
                 print(f'Migrated payments table with {count} existing rows')
         else:
             print('Columns already nullable, no migration needed')
     "
     ```

     ```python
     def format_payment_confirmation(
         paid_seats: List[SeatAssignment],
         remaining_seats: List[SeatAssignment],
         payment_amount: Decimal,
         buyer_display_name: str,
     ) -> str:
         # Group paid seats by row
         paid_by_row = defaultdict(list)
         for seat in paid_seats:
             paid_by_row[seat.row_number].append(seat.seat_number)

         paid_lines = []
         for row in sorted(paid_by_row.keys()):
             seat_nums = sorted(paid_by_row[row])
             paid_lines.append(f"Ряд {row}: места {format_seat_ranges(seat_nums)}")

         result = (
             f"Оплата зарегистрирована для {buyer_display_name}.\n\n"
             f"Оплачено {len(paid_seats)} мест:\n"
             + "\n".join(paid_lines)
             + f"\n\nСумма: {payment_amount} руб."
         )

         if remaining_seats:
             remaining_by_row = defaultdict(list)
             for seat in remaining_seats:
                 remaining_by_row[seat.row_number].append(seat.seat_number)

             remaining_lines = []
             for row in sorted(remaining_by_row.keys()):
                 seat_nums = sorted(remaining_by_row[row])
                 remaining_lines.append(f"Ряд {row}: места {format_seat_ranges(seat_nums)}")

             result += (
                 f"\n\nОсталось к оплате: {len(remaining_seats)} мест:\n"
                 + "\n".join(remaining_lines)
             )
         else:
             result += "\n\nВсе места оплачены!"

         return result
     ```

     This uses `format_seat_ranges` (which formats `[1,2,3,5]` as `"1-3, 5"`) consistently with `format_reservation_seats` in the same file.
