  ```python
  def forwarded_has_receipt_attachment(forwarded) -> bool:
      """Check if forwarded message has photo or document attachment (receipt indicator)."""
      if not hasattr(forwarded, 'attachments') or not forwarded.attachments:
          return False
      for att in forwarded.attachments:
          if att.photo is not None or att.doc is not None:
              return True
      return False
  ```

  Replace ALL instances of `BotState.MAIN_MENU.value` with `str(BotState.MAIN_MENU)`.

  The bug: vkbottle stores state as `str(BotState.MAIN_MENU)` = `"BotState:main_menu"`, but the code compares `current_state == BotState.MAIN_MENU.value` which is `"main_menu"`. These NEVER match, making all MAIN_MENU branches dead code.

  Lines to fix:

  <name>Task 2: Update context_switch_handler to support pending_is_receipt payment flow</name>

  <action>
  Update the context_switch_handler in bot/handlers/menu.py to check for pending_is_receipt flag and call start_payment_flow instead of show_main_menu when confirmed.

  In the `if text == "Да":` block, after `await storage.delete(message.from_id)`:

  **Current code (always shows main menu or creates reservation):**
  ```python
  if pending_has_reservation:
      await show_main_menu(...)
  else:
      # Create new reservation...
  ```

  **Replace with:**
  ```python
  # Extract receipt metadata
  pending_is_receipt = context.get("pending_is_receipt", False)
  pending_receipt_vk_message_ts = context.get("pending_receipt_vk_message_ts")

  if pending_has_reservation:
      if pending_is_receipt:
          # Receipt-triggered switch -> enter payment flow for new client
          # First set MAIN_MENU context so start_payment_flow has correct FSM context
          await storage.set(
              message.from_id,
              BotState.MAIN_MENU,
              reservation_id=pending_reservation_id,
              buyer_vk_user_id=pending_buyer_vk_id,
              buyer_vk_api_id=pending_buyer_vk_api_id,
              buyer_display_name=pending_buyer_display_name,
              from_main_menu=True,
          )
          from bot.handlers.payment import start_payment_flow
          # Parse vk_message_ts from stored ISO string
          vk_message_ts = None
          if pending_receipt_vk_message_ts:
              from datetime import datetime
              try:
                  vk_message_ts = datetime.fromisoformat(pending_receipt_vk_message_ts)
              except (ValueError, TypeError):
                  vk_message_ts = None
          await start_payment_flow(
              message, storage, pending_reservation_id,
              pending_buyer_display_name,
              vk_message_url=None,
              vk_message_ts=vk_message_ts,
          )
      else:
          # Normal switch -> show main menu
          await show_main_menu(
              message,
              storage,
              pending_reservation_id,
              pending_buyer_vk_id,
              pending_buyer_display_name,
              pending_buyer_vk_api_id,
          )
  else:
      # No existing reservation for new client
      if pending_is_receipt:
          # Receipt from client with no reservation — can't record payment, create reservation
          # (edge case: someone forwarded receipt but client has no booking)
          try:
              seller_id = db.get_default_seller_id()
              reservation = db.create_reservation(pending_buyer_vk_id, seller_id)

              await message.answer(
                  f"У {pending_buyer_display_name} нет активной брони.\n"
                  f"Создана новая бронь (ID: {reservation.id}).\n\n"
                  f"Введите номер ряда:",
                  keyboard=get_empty_keyboard()
              )

              await storage.set(
                  message.from_id,
                  BotState.AWAITING_ROW,
                  reservation_id=reservation.id,
                  buyer_vk_user_id=pending_buyer_vk_id,
                  buyer_display_name=pending_buyer_display_name,
                  buyer_vk_api_id=pending_buyer_vk_api_id,
              )
          except db.DatabaseError as e:
              logger.error(f"Error creating reservation on context switch: {e}")
              await message.answer("Ошибка при создании брони. Попробуйте позже.")
      else:
          # Normal switch with no reservation -> create new reservation (existing code)
          try:
              seller_id = db.get_default_seller_id()
              reservation = db.create_reservation(pending_buyer_vk_id, seller_id)

              await message.answer(
                  f"Создана новая бронь для {pending_buyer_display_name} (ID: {reservation.id}).\n\n"
                  f"Введите номер ряда:",
                  keyboard=get_empty_keyboard()
              )

              await storage.set(
                  message.from_id,
                  BotState.AWAITING_ROW,
                  reservation_id=reservation.id,
                  buyer_vk_user_id=pending_buyer_vk_id,
                  buyer_display_name=pending_buyer_display_name,
                  buyer_vk_api_id=pending_buyer_vk_api_id,
              )
          except db.DatabaseError as e:
              logger.error(f"Error creating reservation on context switch: {e}")
              await message.answer("Ошибка при создании брони. Попробуйте позже.")
  ```

  Note: The `await storage.delete(message.from_id)` call should happen BEFORE the receipt/non-receipt branching (it already does). This clears the CONTEXT_SWITCH_CONFIRM state. Then the branches either set MAIN_MENU (for receipt payment flow) or proceed with show_main_menu/create_reservation.
